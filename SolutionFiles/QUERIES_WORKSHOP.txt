CREATE DATABASE OFICINA_DO_LEONIDAS

USE OFICINA_DO_LEONIDAS

CREATE TABLE OFICINA(
    ID_OFICINA INT IDENTITY(1,1) PRIMARY KEY,
    CARGA INT NOT NULL,
    NOME VARCHAR(255) NOT NULL,
    CNPJ VARCHAR(255) NOT NULL,
    SENHA VARCHAR(255) NOT NULL
)

CREATE TABLE CLIENTE(
    ID_CLIENTE INT IDENTITY(1,1) PRIMARY KEY,
    NOME VARCHAR(255) NOT NULL,
    CPF VARCHAR(255) NOT NULL
)

CREATE TABLE SERVICOS(
    ID_SERVICO INT IDENTITY(1,1) PRIMARY KEY,
    DATA DATETIME NOT NULL,
    CARGA INT NOT NULL,
    ID_OFICINA INT NOT NULL,
    ID_CLIENTE INT NOT NULL,
    FOREIGN KEY (ID_OFICINA) REFERENCES Oficina(ID_OFICINA),
    FOREIGN KEY (ID_CLIENTE) REFERENCES Cliente(ID_CLIENTE)
)

CREATE TABLE DIADETRABALHO(
	ID_DIA_DE_TRABALHO INT IDENTITY (1,1) PRIMARY KEY,
	ID_OFICINA INT NOT NULL,
      DATA DATETIME NOT NULL,
      CARGA_DISPONIVEL INT NOT NULL,    
	FOREIGN KEY (ID_OFICINA) REFERENCES Oficina(ID_OFICINA)
)
